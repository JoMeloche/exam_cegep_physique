\subsubsection*{Question a) Énergie potentielle gagnée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m = \qty{\v{m}}{kg}$, $h = \qty{\v{h}}{m}$, $g = \qty{9.8}{N/kg}$ ; variable recherchée : $E_p$
    \item \textbf{Équations pertinentes :} \[ E_p = m g h \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ E_p = \num{\v{m}} \cdot \num{9.8} \cdot \num{\v{h}} = \Res{\vsi{m}*9.8*\vsi{h}}{J} \]
    \item \textbf{Réponse :} $E_p = \mathbf{\Res{\vsi{m}*9.8*\vsi{h}}{J}}$
\end{enumerate}

\subsubsection*{Question b) Puissance utile}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $E_p = \Res{\v{m}*9.8*\v{h}}{J}$, $\Delta t = \qty{\v{t}}{s}$ ; variable recherchée : $P$
    \item \textbf{Équations pertinentes :} \[ P = \frac{E}{\Delta t} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ P = \frac{\Val{\v{m}*9.8*\v{h}}}{\num{\v{t}}} = \Res{\vsi{m}*9.8*\vsi{h}/\vsi{t}}{W} \]
    \item \textbf{Réponse :} $P = \mathbf{\Res{\vsi{m}*9.8*\vsi{h}/\vsi{t}}{W}}$
\end{enumerate}

\subsubsection*{Question c) Énergie dépensée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $E_{\mathrm{utile}} = \Res{\v{m}*9.8*\v{h}}{J}$, rendement $= \qty{25}{\percent}$ ; variable recherchée : $E_{\mathrm{dépensée}}$
    \item \textbf{Équations pertinentes :} \[ \text{rendement} = \frac{E_{\mathrm{utile}}}{E_{\mathrm{dépensée}}} \times 100\,\% \]
    \item \textbf{Isolation de la variable recherchée :} \[ E_{\mathrm{dépensée}} = \frac{E_{\mathrm{utile}}}{0{,}25} \]
    \item \textbf{Résolution numérique :} \[ E_{\mathrm{dépensée}} = \frac{\Val{\v{m}*9.8*\v{h}}}{\num{0.25}} = \Res{\vsi{m}*9.8*\vsi{h}/0.25}{J} \]
    \item \textbf{Réponse :} $E_{\mathrm{dépensée}} = \mathbf{\Res{\vsi{m}*9.8*\vsi{h}/0.25}{J}}$
\end{enumerate}
