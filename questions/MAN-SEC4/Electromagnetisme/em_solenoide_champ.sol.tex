\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\providecommand{\ResSci}[2]{\qty[round-mode=figures,round-precision=3,exponent-mode=scientific]{\fpeval{#1}}{#2}}
\subsubsection*{Question a) Champ magnétique}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $N = \num{\v{N}}$, $L = \qty{\vsi{L}}{m}$, $I = \qty{\v{I}}{A}$, $\mu_0 = \qty{1.257e-6}{T.m/A}$ ; variable recherchée : $B$
    \item \textbf{Équations pertinentes :}
    \[ B = \frac{\mu_0 N I}{L} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ B = \frac{\num{1.257e-6} \cdot \num{\v{N}} \cdot \num{\v{I}}}{\num{\vsi{L}}} = \ResSci{1.257e-6*\v{N}*\v{I}/\vsi{L}}{T} \]
    \item \textbf{Réponse :} $B = \mathbf{\Res{1.257e-3*\v{N}*\v{I}/\vsi{L}}{mT}}$.
\end{enumerate}

\subsubsection*{Question b) Longueur réduite de moitié}
$B$ est inversement proportionnel à $L$ : si $L$ est divisée par 2, $B$ double.
\[ B' = 2B = \Res{2*1.257e-3*\v{N}*\v{I}/\vsi{L}}{mT} \]

\subsubsection*{Question c) Courant nécessaire}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $B = \qty{\vsi{Bc}}{T}$, $N = \num{\v{N}}$, $L = \qty{\vsi{L}}{m}$ ; variable recherchée : $I$
    \item \textbf{Équations pertinentes :}
    \[ B = \frac{\mu_0 N I}{L} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ I = \frac{BL}{\mu_0 N} \]
    \item \textbf{Résolution numérique :}
    \[ I = \frac{\num{\vsi{Bc}} \cdot \num{\vsi{L}}}{\num{1.257e-6} \cdot \num{\v{N}}} = \Res{\vsi{Bc}*\vsi{L}/(1.257e-6*\v{N})}{A} \]
    \item \textbf{Réponse :} il faudrait un courant de \textbf{\Res{\vsi{Bc}*\vsi{L}/(1.257e-6*\v{N})}{A}}.
\end{enumerate}
