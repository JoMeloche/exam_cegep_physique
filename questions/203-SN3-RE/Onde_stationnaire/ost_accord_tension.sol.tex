\subsubsection*{Question a) Tension requise}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $L = \qty{\vsi{L}}{m}$, $m = \qty{\vsi{m}}{kg}$, $\mu = m/L = \Res{\vsi{m}/\vsi{L}}{kg/m}$, $f_1 = \qty{\v{f}}{Hz}$
        \item Variable recherchée : $F$ (N)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ f_1 = \frac{v}{2L} = \frac{1}{2L}\sqrt{\frac{F}{\mu}} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ F = \mu\,(2 L f_1)^2 \]
    \item \textbf{Résolution numérique :}
    \[ F = \Val{\vsi{m}/\vsi{L}} \cdot (2 \cdot \num{\vsi{L}} \cdot \num{\v{f}})^2 = \Res{\vsi{m}/\vsi{L}*(2*\vsi{L}*\v{f})^2}{N} \]
    \item \textbf{Réponse :} il faut une tension de \textbf{\Res{\vsi{m}/\vsi{L}*(2*\vsi{L}*\v{f})^2}{N}}.
\end{enumerate}

\subsubsection*{Question b) Tension initiale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f_1' = \qty{\fpeval{\v{f}-\v{f2}}}{Hz}$, mêmes $L$ et $\mu$ ; variable recherchée : $F'$
    \item \textbf{Équations pertinentes :}
    \[ F' = \mu\,(2 L f_1')^2 \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ F' = \Val{\vsi{m}/\vsi{L}} \cdot (2 \cdot \num{\vsi{L}} \cdot \num{\fpeval{\v{f}-\v{f2}}})^2 = \Res{\vsi{m}/\vsi{L}*(2*\vsi{L}*(\v{f}-\v{f2}))^2}{N} \]
    \item \textbf{Réponse :} la tension initiale était de \textbf{\Res{\vsi{m}/\vsi{L}*(2*\vsi{L}*(\v{f}-\v{f2}))^2}{N}}.
\end{enumerate}

\subsubsection*{Question c) Augmentation relative de la tension}
Puisque $F \propto f_1^2$ :
\[ \frac{F}{F'} - 1 = \left(\frac{\num{\v{f}}}{\num{\fpeval{\v{f}-\v{f2}}}}\right)^2 - 1 = \Res{100*((\v{f}/(\v{f}-\v{f2}))^2 - 1)}{\percent} \]
Il faut augmenter la tension d'environ \textbf{\Res{100*((\v{f}/(\v{f}-\v{f2}))^2 - 1)}{\percent}}.
