\subsubsection*{Question a) Fréquences perçues}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f = \qty{\v{f}}{Hz}$, $v_o = \qty{\v{vo}}{m/s}$, sources immobiles, $v = \qty{\v{v}}{m/s}$ ; variables recherchées : $f_{\text{av}}$, $f_{\text{ar}}$
    \item \textbf{Équations pertinentes :}
    \[ f' = f\,\frac{v \pm v_o}{v} \]
    \item \textbf{Isolation de la variable recherchée :} $+$ pour le haut-parleur dont elle s'approche, $-$ pour l'autre.
    \item \textbf{Résolution numérique :}
    \[ f_{\text{av}} = \num{\v{f}} \cdot \frac{\num{\v{v}} + \num{\v{vo}}}{\num{\v{v}}} = \Res{\v{f}*(\v{v}+\v{vo})/\v{v}}{Hz} \qquad f_{\text{ar}} = \num{\v{f}} \cdot \frac{\num{\v{v}} - \num{\v{vo}}}{\num{\v{v}}} = \Res{\v{f}*(\v{v}-\v{vo})/\v{v}}{Hz} \]
    \item \textbf{Réponse :} elle perçoit \textbf{\Res{\v{f}*(\v{v}+\v{vo})/\v{v}}{Hz}} devant et \textbf{\Res{\v{f}*(\v{v}-\v{vo})/\v{v}}{Hz}} derrière.
\end{enumerate}

\subsubsection*{Question b) Battements}
\[ f_{\text{batt}} = f_{\text{av}} - f_{\text{ar}} = \frac{2 f v_o}{v} = \frac{2 \cdot \num{\v{f}} \cdot \num{\v{vo}}}{\num{\v{v}}} = \Res{2*\v{f}*\v{vo}/\v{v}}{Hz} \]
Elle entend \textbf{\Res{2*\v{f}*\v{vo}/\v{v}}{Hz}} de battements.

\subsubsection*{Question c) Un battement par seconde}
\[ v_o = \frac{f_{\text{batt}}\,v}{2f} = \frac{1 \cdot \num{\v{v}}}{2 \cdot \num{\v{f}}} = \Res{\v{v}/(2*\v{f})}{m/s} \]
Elle devrait marcher à \textbf{\Res{\v{v}/(2*\v{f})}{m/s}}.
