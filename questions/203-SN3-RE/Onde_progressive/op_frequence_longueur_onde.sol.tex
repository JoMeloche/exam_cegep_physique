\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\providecommand{\ResSci}[2]{\qty[round-mode=figures,round-precision=3,exponent-mode=scientific]{\fpeval{#1}}{#2}}
\subsubsection*{Question a) Vitesse de l'onde}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $F = \qty{\v{F}}{N}$, $\mu = \qty{\vsi{mu}}{kg/m}$
        \item Variable recherchée : $v$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v = \sqrt{\frac{F}{\mu}} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ v = \sqrt{\frac{\num{\v{F}}}{\num{\vsi{mu}}}} = \Res{sqrt(\v{F}/\vsi{mu})}{m/s} \]
    \item \textbf{Réponse :} l'onde se propage à \textbf{\Res{sqrt(\v{F}/\vsi{mu})}{m/s}}.
\end{enumerate}

\subsubsection*{Question b) Longueur d'onde}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v = \Res{sqrt(\v{F}/\vsi{mu})}{m/s}$, $f = \qty{\v{f}}{Hz}$
        \item Variable recherchée : $\lambda$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v = \lambda f \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \lambda = \frac{v}{f} \]
    \item \textbf{Résolution numérique :}
    \[ \lambda = \frac{\Val{sqrt(\v{F}/\vsi{mu})}}{\num{\v{f}}} = \Res{sqrt(\v{F}/\vsi{mu})/\v{f}}{m} \]
    \item \textbf{Réponse :} la longueur d'onde est de \textbf{\Res{100*sqrt(\v{F}/\vsi{mu})/\v{f}}{cm}}.
\end{enumerate}

\subsubsection*{Question c) Nouvelle tension}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $\lambda' = \num{\v{n}}\,\lambda$, $f$ et $\mu$ inchangés
        \item Variable recherchée : $F'$ (N)
    \end{itemize}
    \item \textbf{Équations pertinentes :} à fréquence fixe, $\lambda \propto v \propto \sqrt{F}$ :
    \[ \frac{\lambda'}{\lambda} = \sqrt{\frac{F'}{F}} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ F' = F\left(\frac{\lambda'}{\lambda}\right)^2 \]
    \item \textbf{Résolution numérique :}
    \[ F' = \num{\v{F}} \cdot (\num{\v{n}})^2 = \Res{\v{F}*\v{n}^2}{N} \]
    \item \textbf{Réponse :} il faudrait une tension de \textbf{\Res{\v{F}*\v{n}^2}{N}}.
\end{enumerate}
