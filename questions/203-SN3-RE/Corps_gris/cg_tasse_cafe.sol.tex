\subsubsection*{Question a) Puissance nette}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $e = \num{\v{em}}$, $A = \qty{\vsi{S}}{m^2}$, $T = \qty{\fpeval{\v{T}+273}}{K}$, $T_e = \qty{\fpeval{\v{Te}+273}}{K}$ ; variable recherchée : $P_{\text{net}}$
    \item \textbf{Équations pertinentes :}
    \[ P_{\text{net}} = e\sigma A\,(T^4 - T_e^4) \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ P_{\text{net}} = \num{\v{em}} \cdot \num{5.67e-8} \cdot \num{\vsi{S}} \cdot \left((\num{\fpeval{\v{T}+273}})^4 - (\num{\fpeval{\v{Te}+273}})^4\right) = \Res{\v{em}*5.67e-8*\vsi{S}*((\v{T}+273)^4 - (\v{Te}+273)^4)}{W} \]
    \item \textbf{Réponse :} la tasse perd \textbf{\Res{\v{em}*5.67e-8*\vsi{S}*((\v{T}+273)^4 - (\v{Te}+273)^4)}{W}} par rayonnement.
\end{enumerate}

\subsubsection*{Question b) Énergie perdue}
\[ E = P_{\text{net}}\,\Delta t = \Val{\v{em}*5.67e-8*\vsi{S}*((\v{T}+273)^4 - (\v{Te}+273)^4)} \cdot \num{\vsi{t}} = \Res{\v{em}*5.67e-8*\vsi{S}*((\v{T}+273)^4 - (\v{Te}+273)^4)*\vsi{t}}{J} \]

\subsubsection*{Question c) Baisse de température}
\[ \Delta T = \frac{E}{mc} = \frac{\Val{\v{em}*5.67e-8*\vsi{S}*((\v{T}+273)^4 - (\v{Te}+273)^4)*\vsi{t}}}{\num{0.250} \cdot 4186} = \Res{\v{em}*5.67e-8*\vsi{S}*((\v{T}+273)^4 - (\v{Te}+273)^4)*\vsi{t}/(0.25*4186)}{K} \]
Le café se refroidirait d'environ \textbf{\Res{\v{em}*5.67e-8*\vsi{S}*((\v{T}+273)^4 - (\v{Te}+273)^4)*\vsi{t}/(0.25*4186)}{\degreeCelsius}}. En réalité, l'évaporation et la convection de l'air au-dessus du café, ainsi que la conduction vers la table, sont souvent plus importantes que le rayonnement.
