\subsubsection*{Question a) Vitesse de l'avion}
\[ v_s = M v = \num{\v{M}} \cdot \num{\v{v}} = \Res{\v{M}*\v{v}}{m/s} \]
L'avion vole à \textbf{\Res{\v{M}*\v{v}}{m/s}}.

\subsubsection*{Question b) Demi-angle du cône}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $v = \qty{\v{v}}{m/s}$, $v_s = \Res{\v{M}*\v{v}}{m/s}$ ; variable recherchée : $\theta$
    \item \textbf{Équations pertinentes :}
    \[ \sin\theta = \frac{v}{v_s} = \frac{1}{M} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \theta = \arcsin\left(\frac{1}{M}\right) \]
    \item \textbf{Résolution numérique :}
    \[ \theta = \arcsin\left(\frac{1}{\num{\v{M}}}\right) = \Res{asind(1/\v{M})}{\degree} \]
    \item \textbf{Réponse :} le demi-angle du cône est de \textbf{\Res{asind(1/\v{M})}{\degree}}.
\end{enumerate}

\subsubsection*{Question c) Délai du bang}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $h = \qty{\vsi{h}}{m}$, $\theta = \Res{asind(1/\v{M})}{\degree}$, $v_s$ ; variable recherchée : $\Delta t$
    \item \textbf{Équations pertinentes :} le cône atteint l'observateur quand l'avion a parcouru une distance $d$ au-delà de la verticale, avec
    \[ \tan\theta = \frac{h}{d} \qquad d = v_s\,\Delta t \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \Delta t = \frac{h}{v_s\tan\theta} \]
    \item \textbf{Résolution numérique :}
    \[ d = \frac{\num{\vsi{h}}}{\tan(\Val{asind(1/\v{M})}^\circ)} = \Res{\vsi{h}/tand(asind(1/\v{M}))}{m} \qquad \Delta t = \frac{\Val{\vsi{h}/tand(asind(1/\v{M}))}}{\Val{\v{M}*\v{v}}} = \Res{\vsi{h}/tand(asind(1/\v{M}))/(\v{M}*\v{v})}{s} \]
    \item \textbf{Réponse :} l'observateur entend le bang \textbf{\Res{\vsi{h}/tand(asind(1/\v{M}))/(\v{M}*\v{v})}{s}} après le passage de l'avion au-dessus de lui.
\end{enumerate}
