\subsubsection*{Question a) Puissance rayonnée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $r = \qty{\vsi{r}}{m}$, $T = \qty{\v{T}}{K}$, $e = 1$ ; variable recherchée : $P$
    \item \textbf{Équations pertinentes :}
    \[ P = e\sigma A T^4 \qquad A = 4\pi r^2 \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ P = \num{5.67e-8} \cdot 4\pi(\num{\vsi{r}})^2 \cdot (\num{\v{T}})^4 = \Res{5.67e-8*4*pi*\vsi{r}^2*\v{T}^4}{W} \]
    \item \textbf{Réponse :} la sphère rayonne \textbf{\Res{5.67e-8*4*pi*\vsi{r}^2*\v{T}^4}{W}}.
\end{enumerate}

\subsubsection*{Question b) Maximum d'émission}
\[ \lambda_{\max} = \frac{\num{2.898e-3}}{\num{\v{T}}} = \Res{1e6*2.898e-3/\v{T}}{\micro\metre} \quad \text{(infrarouge)} \]

\subsubsection*{Question c) Nombre de photons}
\[ E_\gamma = \frac{hc}{\lambda_{\max}} = \ResSci{6.626e-34*3e8/(2.898e-3/\v{T})}{J} \qquad \frac{N}{\Delta t} = \frac{P}{E_\gamma} = \ResSci{(5.67e-8*4*pi*\vsi{r}^2*\v{T}^4)/(6.626e-34*3e8/(2.898e-3/\v{T}))}{s^{-1}} \]
La sphère émet environ \textbf{\ResSci{(5.67e-8*4*pi*\vsi{r}^2*\v{T}^4)/(6.626e-34*3e8/(2.898e-3/\v{T}))}{}} photons par seconde.
