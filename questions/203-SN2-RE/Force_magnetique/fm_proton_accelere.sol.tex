\subsubsection*{Question a) Vitesse}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $\Delta V = \QtyT{\vsi{dV}}{V}$, $q = e = \qty{1.602e-19}{C}$, $m_p = \qty{1.673e-27}{kg}$ ; variable recherchée : $v$
    \item \textbf{Équations pertinentes :} conservation de l'énergie :
    \[ |q|\,\Delta V = \tfrac{1}{2} m v^2 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ v = \sqrt{\frac{2e\,\Delta V}{m_p}} \]
    \item \textbf{Résolution numérique :}
    \[ v = \sqrt{\frac{2 \cdot \num{1.602e-19} \cdot \NumT{\vsi{dV}}}{\num{1.673e-27}}} = \ResSci{sqrt(2*1.602e-19*\vsi{dV}/1.673e-27)}{m/s} \]
    \item \textbf{Réponse :} $v = \mathbf{\ResSci{sqrt(2*1.602e-19*\vsi{dV}/1.673e-27)}{m/s}}$.
\end{enumerate}

\subsubsection*{Question b) Rayon de la trajectoire}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $v = \ResSci{sqrt(2*1.602e-19*\vsi{dV}/1.673e-27)}{m/s}$, $B = \QtyT{\vsi{B}}{T}$ ; variable recherchée : $r$
    \item \textbf{Équations pertinentes :} la force magnétique joue le rôle de force centripète :
    \[ |q| v B = \frac{m v^2}{r} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ r = \frac{m_p v}{e B} \]
    \item \textbf{Résolution numérique :}
    \[ r = \frac{\num{1.673e-27} \cdot \Valt{sqrt(2*1.602e-19*\vsi{dV}/1.673e-27)}}{\num{1.602e-19} \cdot \NumT{\vsi{B}}} = \Res{1.673e-27*(sqrt(2*1.602e-19*\vsi{dV}/1.673e-27))/(1.602e-19*\vsi{B})}{m} \]
    \item \textbf{Réponse :} $r = \mathbf{\Res{100*1.673e-27*(sqrt(2*1.602e-19*\vsi{dV}/1.673e-27))/(1.602e-19*\vsi{B})}{cm}}$.
\end{enumerate}

\subsubsection*{Question c) Période}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $B = \QtyT{\vsi{B}}{T}$, $m_p$, $e$ ; variable recherchée : $T$
    \item \textbf{Équations pertinentes :}
    \[ T = \frac{2\pi r}{v} \qquad r = \frac{m_p v}{eB} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ T = \frac{2\pi m_p}{eB} \]
    \item \textbf{Résolution numérique :}
    \[ T = \frac{2\pi \cdot \num{1.673e-27}}{\num{1.602e-19} \cdot \NumT{\vsi{B}}} = \ResSci{2*pi*1.673e-27/(1.602e-19*\vsi{B})}{s} \]
    \item \textbf{Réponse :} $T = \mathbf{\ResSci{2*pi*1.673e-27/(1.602e-19*\vsi{B})}{s}}$ ; elle ne dépend pas de la vitesse.
\end{enumerate}
