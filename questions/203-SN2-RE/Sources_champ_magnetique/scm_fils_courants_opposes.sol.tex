\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\providecommand{\ResSci}[2]{\qty[round-mode=figures,round-precision=3,exponent-mode=scientific]{\fpeval{#1}}{#2}}

\subsubsection*{Question a) Champ d'un fil au milieu}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $I = \qty{\v{I}}{A}$, $r = d/2 = \qty{\fpeval{\vsi{d}/2}}{m}$, $\mu_0 = 4\pi \times \qty{e-7}{T.m/A}$ ; variable recherchée : $B_1$
    \item \textbf{Équations pertinentes :} \[ B = \frac{\mu_0 I}{2\pi r} \]
    \item \textbf{Isolation de la variable recherchée :} \[ B_1 = \frac{\mu_0 I}{\pi d} \]
    \item \textbf{Résolution numérique :} \[ B_1 = \frac{4\pi\times 10^{-7} \cdot \num{\v{I}}}{\pi \cdot \num{\vsi{d}}} = \ResSci{(4*pi*1e-7)*\vsi{I}/(2*pi*\vsi{d}/2)}{T} \]
    \item \textbf{Réponse :} $B_1 = \mathbf{\ResSci{(4*pi*1e-7)*\vsi{I}/(2*pi*\vsi{d}/2)}{T}}$
\end{enumerate}

\subsubsection*{Question b) Champ résultant au milieu}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $B_1 = \ResSci{4e-7*\v{I}/\vsi{d}}{T}$ pour chaque fil ; variable recherchée : $B$
    \item \textbf{Équations pertinentes :} avec des courants opposés, les deux champs ont le même sens entre les fils (règle de la main droite) :\[ B = B_1 + B_2 = 2 B_1 \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ B = 2 \cdot \ResSci{4e-7*\v{I}/\vsi{d}}{T} = \ResSci{2*(4*pi*1e-7)*\vsi{I}/(2*pi*\vsi{d}/2)}{T} \]
    \item \textbf{Réponse :} $B = \mathbf{\ResSci{2*(4*pi*1e-7)*\vsi{I}/(2*pi*\vsi{d}/2)}{T}}$
\end{enumerate}

\subsubsection*{Question c) Force entre les fils}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $I_1 = I_2 = \qty{\v{I}}{A}$, $d = \qty{\vsi{d}}{m}$, $L = \qty{\v{L}}{m}$ ; variable recherchée : $F$
    \item \textbf{Équations pertinentes :} \[ F = I_2 L B_{1\to 2} = \frac{\mu_0 I_1 I_2 L}{2\pi d} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ F = \frac{4\pi\times 10^{-7} \cdot (\num{\v{I}})^2 \cdot \num{\v{L}}}{2\pi \cdot \num{\vsi{d}}} = \ResSci{(4*pi*1e-7)*\vsi{I}^2*\vsi{L}/(2*pi*\vsi{d})}{N} \]
    \item \textbf{Réponse :} $F = \mathbf{\ResSci{(4*pi*1e-7)*\vsi{I}^2*\vsi{L}/(2*pi*\vsi{d})}{N}}$ ; des courants opposés se \textbf{repoussent}.
\end{enumerate}
