\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\providecommand{\ResSci}[2]{\qty[round-mode=figures,round-precision=3,exponent-mode=scientific]{\fpeval{#1}}{#2}}
\subsubsection*{Question a) Champ électrique}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $\Delta V = \qty{\v{dV}}{V}$, $d = \qty{\vsi{d}}{m}$ ; variable recherchée : $E$
    \item \textbf{Équations pertinentes :} champ uniforme :
    \[ \Delta V = Ed \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ E = \frac{\Delta V}{d} \]
    \item \textbf{Résolution numérique :}
    \[ E = \frac{\num{\v{dV}}}{\num{\vsi{d}}} = \Res{\v{dV}/\vsi{d}}{V/m} \]
    \item \textbf{Réponse :} $E = \mathbf{\Res{\v{dV}/\vsi{d}}{V/m}}$, de la plaque positive vers la plaque négative.
\end{enumerate}

\subsubsection*{Question b) Force sur le proton}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $E$, $e = \qty{1.602e-19}{C}$ ; variable recherchée : $F$
    \item \textbf{Équations pertinentes :}
    \[ F = eE \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ F = \num{1.602e-19} \cdot \Val{\v{dV}/\vsi{d}} = \ResSci{1.602e-19*\v{dV}/\vsi{d}}{N} \]
    \item \textbf{Réponse :} $F = \mathbf{\ResSci{1.602e-19*\v{dV}/\vsi{d}}{N}}$, vers la plaque négative.
\end{enumerate}

\subsubsection*{Question c) Vitesse finale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $\Delta V$, $m_p = \qty{1.673e-27}{kg}$, $v_i = 0$ ; variable recherchée : $v$
    \item \textbf{Équations pertinentes :} conservation de l'énergie :
    \[ e\Delta V = \frac{1}{2}m_pv^2 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ v = \sqrt{\frac{2e\Delta V}{m_p}} \]
    \item \textbf{Résolution numérique :}
    \[ v = \sqrt{\frac{2 \cdot \num{1.602e-19} \cdot \num{\v{dV}}}{\num{1.673e-27}}} = \ResSci{sqrt(2*1.602e-19*\v{dV}/1.673e-27)}{m/s} \]
    \item \textbf{Réponse :} $v = \mathbf{\ResSci{sqrt(2*1.602e-19*\v{dV}/1.673e-27)}{m/s}}$ (indépendante de $d$).
\end{enumerate}
