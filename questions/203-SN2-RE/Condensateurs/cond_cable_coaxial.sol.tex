\subsubsection*{Question a) Capacité}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $L = \qty{\v{L}}{m}$, $a = \qty{\vsi{ra}}{m}$, $b = \qty{\vsi{rb}}{m}$, $\varepsilon_0 = \qty{8.85e-12}{F/m}$ ; variable recherchée : $C$
    \item \textbf{Équations pertinentes :} \[ C = \frac{2\pi\varepsilon_0 L}{\ln(b/a)} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ C = \frac{2\pi \cdot \num{8.85e-12} \cdot \num{\v{L}}}{\ln(\num{\v{rb}}/\num{\v{ra}})} = \ResSci{2*pi*8.85e-12*\vsi{L}/ln(\vsi{rb}/\vsi{ra})}{F} \]
    \item \textbf{Réponse :} $C = \mathbf{\ResSci{2*pi*8.85e-12*\vsi{L}/ln(\vsi{rb}/\vsi{ra})}{F}}$
\end{enumerate}

\subsubsection*{Question b) Charge du conducteur central}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $C = \ResSci{2*pi*8.85e-12*\v{L}/ln(\v{rb}/\v{ra})}{F}$, $\Delta V = \qty{\v{V}}{V}$ ; variable recherchée : $Q$
    \item \textbf{Équations pertinentes :} \[ Q = C\,\Delta V \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ Q = \ResSci{2*pi*8.85e-12*\v{L}/ln(\v{rb}/\v{ra})}{F} \cdot \qty{\v{V}}{V} = \ResSci{2*pi*8.85e-12*\vsi{L}/ln(\vsi{rb}/\vsi{ra})*\vsi{V}}{C} \]
    \item \textbf{Réponse :} $Q = \mathbf{\ResSci{2*pi*8.85e-12*\vsi{L}/ln(\vsi{rb}/\vsi{ra})*\vsi{V}}{C}}$
\end{enumerate}

\subsubsection*{Question c) Énergie emmagasinée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $C = \ResSci{2*pi*8.85e-12*\v{L}/ln(\v{rb}/\v{ra})}{F}$, $\Delta V = \qty{\v{V}}{V}$ ; variable recherchée : $U$
    \item \textbf{Équations pertinentes :} \[ U = \tfrac{1}{2} C\,(\Delta V)^2 \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ U = \tfrac{1}{2} \cdot \ResSci{2*pi*8.85e-12*\v{L}/ln(\v{rb}/\v{ra})}{F} \cdot (\qty{\v{V}}{V})^2 = \ResSci{pi*8.85e-12*\vsi{L}/ln(\vsi{rb}/\vsi{ra})*\vsi{V}^2}{J} \]
    \item \textbf{Réponse :} $U = \mathbf{\ResSci{pi*8.85e-12*\vsi{L}/ln(\vsi{rb}/\vsi{ra})*\vsi{V}^2}{J}}$
\end{enumerate}
