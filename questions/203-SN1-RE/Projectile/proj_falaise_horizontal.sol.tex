\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Temps de vol}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Hauteur de la falaise : $h = \qty{\v{h}}{m}$
        \item $v_{0y} = \qty{0}{m/s}$, $a_y = -g$
        \item Variable recherchée : $t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ y = y_0 + v_{0y} t - \frac{1}{2} g t^2 \implies 0 = h - \frac{1}{2} g t^2 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ t = \sqrt{\frac{2h}{g}} \]
    \item \textbf{Résolution numérique :}
    \[ t = \sqrt{\frac{2 \cdot \num{\vsi{h}}}{\num{9.81}}} = \Res{sqrt(2*\vsi{h}/9.81)}{s} \]
    \item \textbf{Réponse :} la pierre vole pendant \textbf{\Res{sqrt(2*\vsi{h}/9.81)}{s}}.
\end{enumerate}

\subsubsection*{Question b) Portée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Vitesse horizontale initiale : $v_{0x} = \qty{\v{v0}}{m/s}$
        \item $t = \Res{sqrt(2*\vsi{h}/9.81)}{s}$
        \item Variable recherchée : $x$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ x = v_{0x} t \quad (a_x = 0) \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ x = \num{\vsi{v0}} \cdot \Val{sqrt(2*\vsi{h}/9.81)} = \Res{\vsi{v0}*sqrt(2*\vsi{h}/9.81)}{m} \]
    \item \textbf{Réponse :} la pierre tombe à \textbf{\Res{\vsi{v0}*sqrt(2*\vsi{h}/9.81)}{m}} du pied de la falaise.
\end{enumerate}

\subsubsection*{Question c) Vitesse à l'impact}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Vitesse horizontale initiale : $v_x = \qty{\v{v0}}{m/s}$
        \item Variable recherchée : $v$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v_y^2 = v_{0y}^2 + 2 g h \qquad v = \sqrt{v_x^2 + v_y^2} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ v = \sqrt{v_0^2 + 2 g h} \]
    \item \textbf{Résolution numérique :}
    \[ |v_y| = \Res{sqrt(2*9.81*\vsi{h})}{m/s} \implies v = \sqrt{(\num{\vsi{v0}})^2 + (\Val{sqrt(2*9.81*\vsi{h})})^2} = \Res{sqrt(\vsi{v0}^2 + 2*9.81*\vsi{h})}{m/s} \]
    \item \textbf{Réponse :} la pierre frappe le sol à \textbf{\Res{sqrt(\vsi{v0}^2 + 2*9.81*\vsi{h})}{m/s}}.
\end{enumerate}

\subsubsection*{Question d) Angle à l'impact}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_x = \qty{\v{v0}}{m/s}$, $|v_y| = \Res{sqrt(2*9.81*\vsi{h})}{m/s}$
        \item Variable recherchée : $\theta$ (sous l'horizontale)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ \tan\theta = \frac{|v_y|}{v_x} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \theta = \arctan\left(\frac{|v_y|}{v_x}\right) \]
    \item \textbf{Résolution numérique :}
    \[ \theta = \arctan\left(\frac{\Val{sqrt(2*9.81*\vsi{h})}}{\num{\vsi{v0}}}\right) = \Res{atand(sqrt(2*9.81*\vsi{h})/\vsi{v0})}{\degree} \]
    \item \textbf{Réponse :} la vitesse pointe à \textbf{\Res{atand(sqrt(2*9.81*\vsi{h})/\vsi{v0})}{\degree}} sous l'horizontale.
\end{enumerate}
