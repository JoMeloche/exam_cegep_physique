\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Durée de la chute}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Altitude de l'avion : $h = \qty{\v{h}}{m}$
        \item $v_{0y} = 0$ (le colis a la vitesse horizontale de l'avion)
        \item Variable recherchée : $t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ 0 = h - \frac{1}{2} g t^2 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ t = \sqrt{\frac{2h}{g}} \]
    \item \textbf{Résolution numérique :}
    \[ t = \sqrt{\frac{2 \cdot \num{\vsi{h}}}{\num{9.81}}} = \Res{sqrt(2*\vsi{h}/9.81)}{s} \]
    \item \textbf{Réponse :} la chute dure \textbf{\Res{sqrt(2*\vsi{h}/9.81)}{s}}.
\end{enumerate}

\subsubsection*{Question b) Distance de largage}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Vitesse de l'avion : $v_{0x} = \qty{\v{vk}}{km/h}$ $= \qty[round-mode=figures,round-precision=4,drop-zero-decimal]{\vsi{vk}}{m/s}$
        \item $t = \Res{sqrt(2*\vsi{h}/9.81)}{s}$
        \item Variable recherchée : $x$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ x = v_{0x} t \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ x = \Val{\vsi{vk}} \cdot \Val{sqrt(2*\vsi{h}/9.81)} = \Res{\vsi{vk}*sqrt(2*\vsi{h}/9.81)}{m} \]
    \item \textbf{Réponse :} il faut larguer le colis \textbf{\Res{\vsi{vk}*sqrt(2*\vsi{h}/9.81)}{m}} avant de survoler l'île.
\end{enumerate}

\subsubsection*{Question c) Vitesse à l'impact}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_x = \Res{\vsi{vk}}{m/s}$
        \item $|v_y| = \sqrt{2gh} = \Res{sqrt(2*9.81*\vsi{h})}{m/s}$
        \item Variable recherchée : $v$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v = \sqrt{v_x^2 + v_y^2} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ v = \sqrt{(\Val{\vsi{vk}})^2 + (\Val{sqrt(2*9.81*\vsi{h})})^2} = \Res{sqrt(\vsi{vk}^2 + 2*9.81*\vsi{h})}{m/s} \]
    \item \textbf{Réponse :} le colis touche le sol à \textbf{\Res{sqrt(\vsi{vk}^2 + 2*9.81*\vsi{h})}{m/s}}.
\end{enumerate}

\subsubsection*{Question d) Position de l'avion}
Le colis et l'avion ont la même vitesse horizontale et aucune accélération horizontale (on néglige l'air) : l'avion se trouve donc \textbf{directement au-dessus du colis} au moment de l'impact.
