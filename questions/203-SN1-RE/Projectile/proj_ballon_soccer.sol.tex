\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Hauteur maximale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Vitesse initiale : $v_0 = \qty{\v{v0}}{m/s}$
        \item Angle de lancement : $\theta = \qty{\v{th}}{\degree}$
        \item $v_{0y} = v_0 \sin\theta = \Res{\vsi{v0}*sind(\vsi{th})}{m/s}$
        \item Au sommet : $v_y = 0$
        \item Variable recherchée : $y_{\max}$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v_y^2 = v_{0y}^2 - 2 g y \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ y_{\max} = \frac{v_{0y}^2}{2g} \]
    \item \textbf{Résolution numérique :}
    \[ y_{\max} = \frac{(\Val{\vsi{v0}*sind(\vsi{th})})^2}{2 \cdot \num{9.81}} = \Res{(\vsi{v0}*sind(\vsi{th}))^2/(2*9.81)}{m} \]
    \item \textbf{Réponse :} le ballon monte jusqu'à \textbf{\Res{(\vsi{v0}*sind(\vsi{th}))^2/(2*9.81)}{m}}.
\end{enumerate}

\subsubsection*{Question b) Temps de vol}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_{0y} = \Res{\vsi{v0}*sind(\vsi{th})}{m/s}$
        \item Retour au sol : $y = 0$
        \item Variable recherchée : $t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ y = v_{0y} t - \frac{1}{2} g t^2 = 0 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ t = \frac{2 v_{0y}}{g} \quad (t \neq 0) \]
    \item \textbf{Résolution numérique :}
    \[ t = \frac{2 \cdot \Val{\vsi{v0}*sind(\vsi{th})}}{\num{9.81}} = \Res{2*\vsi{v0}*sind(\vsi{th})/9.81}{s} \]
    \item \textbf{Réponse :} le ballon reste \textbf{\Res{2*\vsi{v0}*sind(\vsi{th})/9.81}{s}} dans les airs.
\end{enumerate}

\subsubsection*{Question c) Portée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_{0x} = v_0\cos\theta = \Res{\vsi{v0}*cosd(\vsi{th})}{m/s}$
        \item $t = \Res{2*\vsi{v0}*sind(\vsi{th})/9.81}{s}$
        \item Variable recherchée : $x$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ x = v_{0x} t \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ x = v_0\cos\theta \cdot \frac{2 v_0 \sin\theta}{g} = \frac{v_0^2 \sin(2\theta)}{g} \]
    \item \textbf{Résolution numérique :}
    \[ x = \Val{\vsi{v0}*cosd(\vsi{th})} \cdot \Val{2*\vsi{v0}*sind(\vsi{th})/9.81} = \Res{\vsi{v0}^2*sind(2*\vsi{th})/9.81}{m} \]
    \item \textbf{Réponse :} le ballon retombe à \textbf{\Res{\vsi{v0}^2*sind(2*\vsi{th})/9.81}{m}} de la joueuse.
\end{enumerate}
