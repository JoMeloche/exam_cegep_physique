\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Hauteur maximale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Hauteur de l'édifice : $y_0 = \qty{\v{h}}{m}$
        \item $v_{0y} = v_0\sin\theta = \Res{\vsi{v0}*sind(\vsi{th})}{m/s}$
        \item Variable recherchée : $y_{\max}$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v_y^2 = v_{0y}^2 - 2 g (y - y_0) \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ y_{\max} = y_0 + \frac{v_{0y}^2}{2g} \]
    \item \textbf{Résolution numérique :}
    \[ y_{\max} = \num{\vsi{h}} + \frac{(\Val{\vsi{v0}*sind(\vsi{th})})^2}{2\cdot\num{9.81}} = \Res{\vsi{h} + (\vsi{v0}*sind(\vsi{th}))^2/(2*9.81)}{m} \]
    \item \textbf{Réponse :} la balle monte jusqu'à \textbf{\Res{\vsi{h} + (\vsi{v0}*sind(\vsi{th}))^2/(2*9.81)}{m}} au-dessus du sol.
\end{enumerate}

\subsubsection*{Question b) Temps de vol}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Hauteur de l'édifice : $y_0 = \qty{\v{h}}{m}$
        \item $v_{0y} = \Res{\vsi{v0}*sind(\vsi{th})}{m/s}$
        \item Au sol : $y = 0$
        \item Variable recherchée : $t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ 0 = y_0 + v_{0y} t - \frac{1}{2} g t^2 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ t = \frac{v_{0y} + \sqrt{v_{0y}^2 + 2 g y_0}}{g} \quad \text{(racine positive)} \]
    \item \textbf{Résolution numérique :}
    \[ t = \frac{\Val{\vsi{v0}*sind(\vsi{th})} + \sqrt{(\Val{\vsi{v0}*sind(\vsi{th})})^2 + 2\cdot\num{9.81}\cdot\num{\vsi{h}}}}{\num{9.81}} = \Res{(\vsi{v0}*sind(\vsi{th}) + sqrt((\vsi{v0}*sind(\vsi{th}))^2 + 2*9.81*\vsi{h}))/9.81}{s} \]
    \item \textbf{Réponse :} la balle touche le sol \textbf{\Res{(\vsi{v0}*sind(\vsi{th}) + sqrt((\vsi{v0}*sind(\vsi{th}))^2 + 2*9.81*\vsi{h}))/9.81}{s}} après le lancer.
\end{enumerate}

\subsubsection*{Question c) Portée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_{0x} = v_0\cos\theta = \Res{\vsi{v0}*cosd(\vsi{th})}{m/s}$
        \item $t = \Res{(\vsi{v0}*sind(\vsi{th}) + sqrt((\vsi{v0}*sind(\vsi{th}))^2 + 2*9.81*\vsi{h}))/9.81}{s}$
        \item Variable recherchée : $x$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ x = v_{0x} t \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ x = \Val{\vsi{v0}*cosd(\vsi{th})} \cdot \Val{(\vsi{v0}*sind(\vsi{th}) + sqrt((\vsi{v0}*sind(\vsi{th}))^2 + 2*9.81*\vsi{h}))/9.81} = \Res{\vsi{v0}*cosd(\vsi{th})*(\vsi{v0}*sind(\vsi{th}) + sqrt((\vsi{v0}*sind(\vsi{th}))^2 + 2*9.81*\vsi{h}))/9.81}{m} \]
    \item \textbf{Réponse :} la balle touche le sol à \textbf{\Res{\vsi{v0}*cosd(\vsi{th})*(\vsi{v0}*sind(\vsi{th}) + sqrt((\vsi{v0}*sind(\vsi{th}))^2 + 2*9.81*\vsi{h}))/9.81}{m}} de l'édifice.
\end{enumerate}
