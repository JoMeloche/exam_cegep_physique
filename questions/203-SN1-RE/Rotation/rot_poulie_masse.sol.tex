\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}

\subsubsection*{Question a) Accélération de la masse}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m = \qty{\v{m}}{kg}$, $M = \qty{\v{M}}{kg}$, $R = \qty{\vsi{R}}{m}$ ; variable recherchée : $a$
    \item \textbf{Équations pertinentes :} masse (axe vers le bas) et poulie, avec $a = \alpha R$ : \[ mg - T = ma \qquad TR = \tfrac{1}{2}MR^2\alpha \implies T = \tfrac{1}{2}Ma \]
    \item \textbf{Isolation de la variable recherchée :} en additionnant : \[ a = \frac{mg}{m + M/2} \]
    \item \textbf{Résolution numérique :} \[ a = \frac{\num{\v{m}} \cdot \num{9.81}}{\num{\v{m}} + \num{\v{M}}/2} = \Res{(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))}{m/s^2} \]
    \item \textbf{Réponse :} $a = \mathbf{\Res{(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))}{m/s^2}}$
\end{enumerate}

\subsubsection*{Question b) Tension}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m = \qty{\v{m}}{kg}$, $a = \Res{(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))}{m/s^2}$ ; variable recherchée : $T$
    \item \textbf{Équations pertinentes :} \[ mg - T = ma \]
    \item \textbf{Isolation de la variable recherchée :} \[ T = m(g - a) \]
    \item \textbf{Résolution numérique :} \[ T = \num{\v{m}} \cdot (\num{9.81} - \Val{(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))}) = \Res{(\vsi{m}*(9.81 - (\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))))}{N} \]
    \item \textbf{Réponse :} $T = \mathbf{\Res{(\vsi{m}*(9.81 - (\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))))}{N}}$ (vérification : $T = \tfrac{1}{2}Ma = \Res{\vsi{M}*((\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2)))/2}{N}$).
\end{enumerate}

\subsubsection*{Question c) Vitesse angulaire finale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $a = \Res{(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))}{m/s^2}$, $h = \qty{\v{h}}{m}$, $v_i = \qty{0}{m/s}$, $R = \qty{\vsi{R}}{m}$ ; variable recherchée : $\omega$
    \item \textbf{Équations pertinentes :} \[ v^2 = v_i^2 + 2ah \qquad v = \omega R \]
    \item \textbf{Isolation de la variable recherchée :} \[ \omega = \frac{\sqrt{2ah}}{R} \]
    \item \textbf{Résolution numérique :} \[ v = \sqrt{2 \cdot \Val{(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))} \cdot \num{\v{h}}} = \Res{sqrt(2*(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))*\vsi{h})}{m/s} \qquad \omega = \frac{\Val{sqrt(2*(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))*\vsi{h})}}{\num{\vsi{R}}} = \Res{(sqrt(2*(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))*\vsi{h})/\vsi{R})}{rad/s} \]
    \item \textbf{Réponse :} $\omega = \mathbf{\Res{(sqrt(2*(\vsi{m}*9.81/(\vsi{m} + \vsi{M}/2))*\vsi{h})/\vsi{R})}{rad/s}}$
\end{enumerate}
