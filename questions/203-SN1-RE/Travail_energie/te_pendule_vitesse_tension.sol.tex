\subsubsection*{Question a) Vitesse au point le plus bas}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $L = \qty{\v{L}}{m}$, $\theta_0 = \qty{\v{th0}}{\degree}$, $v_i = \qty{0}{m/s}$ ; variable recherchée : $v$
    \item \textbf{Équations pertinentes :} la masse descend de $h = L(1 - \cos\theta_0)$ et la tension ne travaille pas : \[ mgL(1-\cos\theta_0) = \tfrac{1}{2}mv^2 \]
    \item \textbf{Isolation de la variable recherchée :} \[ v = \sqrt{2gL(1-\cos\theta_0)} \]
    \item \textbf{Résolution numérique :} \[ v = \sqrt{2 \cdot \num{9.81} \cdot \num{\v{L}} \cdot (1 - \cos(\qty{\v{th0}}{\degree}))} = \Res{(sqrt(2*9.81*\vsi{L}*(1 - cosd(\vsi{th0}))))}{m/s} \]
    \item \textbf{Réponse :} $v = \mathbf{\Res{(sqrt(2*9.81*\vsi{L}*(1 - cosd(\vsi{th0}))))}{m/s}}$
\end{enumerate}

\subsubsection*{Question b) Tension au point le plus bas}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m = \qty{\v{m}}{kg}$, $v = \Res{(sqrt(2*9.81*\vsi{L}*(1 - cosd(\vsi{th0}))))}{m/s}$, $r = L = \qty{\v{L}}{m}$ ; variable recherchée : $T$
    \item \textbf{Équations pertinentes :} deuxième loi de Newton selon la direction centripète (vers le haut) : \[ T - mg = \frac{mv^2}{L} \]
    \item \textbf{Isolation de la variable recherchée :} \[ T = mg + \frac{mv^2}{L} \]
    \item \textbf{Résolution numérique :} \[ T = \num{\v{m}} \cdot \num{9.81} + \frac{\num{\v{m}} \cdot \left(\Val{(sqrt(2*9.81*\vsi{L}*(1 - cosd(\vsi{th0}))))}\right)^2}{\num{\v{L}}} = \Res{(\vsi{m}*9.81*(3 - 2*cosd(\vsi{th0})))}{N} \]
    \item \textbf{Réponse :} $T = \mathbf{\Res{(\vsi{m}*9.81*(3 - 2*cosd(\vsi{th0})))}{N}}$
\end{enumerate}

\subsubsection*{Question c) Vitesse à l'angle intermédiaire}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $\theta_0 = \qty{\v{th0}}{\degree}$, $\theta_1 = \qty{\v{th1}}{\degree}$, $L = \qty{\v{L}}{m}$ ; variable recherchée : $v_1$
    \item \textbf{Équations pertinentes :} entre $\theta_0$ et $\theta_1$, la masse descend de $L(\cos\theta_1 - \cos\theta_0)$ : \[ mgL(\cos\theta_1 - \cos\theta_0) = \tfrac{1}{2}mv_1^2 \]
    \item \textbf{Isolation de la variable recherchée :} \[ v_1 = \sqrt{2gL(\cos\theta_1 - \cos\theta_0)} \]
    \item \textbf{Résolution numérique :} \[ v_1 = \sqrt{2 \cdot \num{9.81} \cdot \num{\v{L}} \cdot (\cos(\qty{\v{th1}}{\degree}) - \cos(\qty{\v{th0}}{\degree}))} = \Res{(sqrt(2*9.81*\vsi{L}*(cosd(\vsi{th1}) - cosd(\vsi{th0}))))}{m/s} \]
    \item \textbf{Réponse :} $v_1 = \mathbf{\Res{(sqrt(2*9.81*\vsi{L}*(cosd(\vsi{th1}) - cosd(\vsi{th0}))))}{m/s}}$
\end{enumerate}
