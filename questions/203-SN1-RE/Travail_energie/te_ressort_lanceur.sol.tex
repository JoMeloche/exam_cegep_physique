\subsubsection*{Question a) Énergie élastique}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $k = \qty{\v{k}}{N/m}$, $x = \qty{\vsi{x}}{m}$ ; variable recherchée : $U_r$
    \item \textbf{Équations pertinentes :} \[ U_r = \tfrac{1}{2}kx^2 \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ U_r = \tfrac{1}{2} \cdot \num{\v{k}} \cdot \left(\num{\vsi{x}}\right)^2 = \Res{(0.5*\vsi{k}*\vsi{x}^2)}{J} \]
    \item \textbf{Réponse :} $U_r = \mathbf{\Res{(0.5*\vsi{k}*\vsi{x}^2)}{J}}$
\end{enumerate}

\subsubsection*{Question b) Vitesse en quittant le ressort}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $U_r = \Res{(0.5*\vsi{k}*\vsi{x}^2)}{J}$, $m = \qty{\vsi{m}}{kg}$, montée de $x = \qty{\vsi{x}}{m}$ ; variable recherchée : $v$
    \item \textbf{Équations pertinentes :} conservation de l'énergie entre le départ et la longueur naturelle : \[ \tfrac{1}{2}kx^2 = \tfrac{1}{2}mv^2 + mgx \]
    \item \textbf{Isolation de la variable recherchée :} \[ v = \sqrt{\frac{kx^2}{m} - 2gx} \]
    \item \textbf{Résolution numérique :} \[ v = \sqrt{\frac{\num{\v{k}} \cdot \left(\num{\vsi{x}}\right)^2}{\num{\vsi{m}}} - 2 \cdot \num{9.81} \cdot \num{\vsi{x}}} = \Res{(sqrt(\vsi{k}*\vsi{x}^2/\vsi{m} - 2*9.81*\vsi{x}))}{m/s} \]
    \item \textbf{Réponse :} $v = \mathbf{\Res{(sqrt(\vsi{k}*\vsi{x}^2/\vsi{m} - 2*9.81*\vsi{x}))}{m/s}}$
\end{enumerate}

\subsubsection*{Question c) Hauteur maximale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $U_r = \Res{(0.5*\vsi{k}*\vsi{x}^2)}{J}$, $m = \qty{\vsi{m}}{kg}$, $v_{\text{haut}} = \qty{0}{m/s}$ ; variable recherchée : $h$
    \item \textbf{Équations pertinentes :} conservation de l'énergie entre le départ et le point le plus haut : \[ \tfrac{1}{2}kx^2 = mgh \]
    \item \textbf{Isolation de la variable recherchée :} \[ h = \frac{kx^2}{2mg} \]
    \item \textbf{Résolution numérique :} \[ h = \frac{\Val{(0.5*\vsi{k}*\vsi{x}^2)}}{\num{\vsi{m}} \cdot \num{9.81}} = \Res{(\vsi{k}*\vsi{x}^2/(2*\vsi{m}*9.81))}{m} \]
    \item \textbf{Réponse :} $h = \mathbf{\Res{(\vsi{k}*\vsi{x}^2/(2*\vsi{m}*9.81))}{m}}$ au-dessus du point de départ.
\end{enumerate}
