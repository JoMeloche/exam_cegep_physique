\subsubsection*{Question a) Travail du moteur}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m = \qty{\v{m}}{kg}$, $h = \qty{\v{h}}{m}$, vitesse constante ; variable recherchée : $W_{\text{mot}}$
    \item \textbf{Équations pertinentes :} à vitesse constante, $\Delta K = 0$, donc $W_{\text{mot}} + W_g = 0$ avec $W_g = -mgh$.
    \item \textbf{Isolation de la variable recherchée :} \[ W_{\text{mot}} = mgh \]
    \item \textbf{Résolution numérique :} \[ W_{\text{mot}} = \num{\v{m}} \cdot \num{9.81} \cdot \num{\v{h}} = \Res{(\vsi{m}*9.81*\vsi{h})}{J} \]
    \item \textbf{Réponse :} $W_{\text{mot}} = \mathbf{\Res{(\vsi{m}*9.81*\vsi{h})}{J}}$
\end{enumerate}

\subsubsection*{Question b) Durée de la montée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $h = \qty{\v{h}}{m}$, $v = \qty{\v{v}}{m/s}$ ; variable recherchée : $\Delta t$
    \item \textbf{Équations pertinentes :} \[ v = \frac{h}{\Delta t} \]
    \item \textbf{Isolation de la variable recherchée :} \[ \Delta t = \frac{h}{v} \]
    \item \textbf{Résolution numérique :} \[ \Delta t = \frac{\num{\v{h}}}{\num{\v{v}}} = \Res{(\vsi{h}/\vsi{v})}{s} \]
    \item \textbf{Réponse :} $\Delta t = \mathbf{\Res{(\vsi{h}/\vsi{v})}{s}}$
\end{enumerate}

\subsubsection*{Question c) Puissance mécanique}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $W_{\text{mot}} = \Res{(\vsi{m}*9.81*\vsi{h})}{J}$, $\Delta t = \Res{(\vsi{h}/\vsi{v})}{s}$ ; variable recherchée : $P_{\text{méc}}$
    \item \textbf{Équations pertinentes :} \[ P = \frac{W}{\Delta t} = Fv = mgv \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :} \[ P_{\text{méc}} = \num{\v{m}} \cdot \num{9.81} \cdot \num{\v{v}} = \Res{(\vsi{m}*9.81*\vsi{v})}{W} \]
    \item \textbf{Réponse :} $P_{\text{méc}} = \mathbf{\Res{(\vsi{m}*9.81*\vsi{v})}{W}}$
\end{enumerate}

\subsubsection*{Question d) Puissance électrique}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $P_{\text{méc}} = \Res{(\vsi{m}*9.81*\vsi{v})}{W}$, $\eta = \num{\vsi{eta}}$ ; variable recherchée : $P_{\text{élec}}$
    \item \textbf{Équations pertinentes :} \[ \eta = \frac{P_{\text{méc}}}{P_{\text{élec}}} \]
    \item \textbf{Isolation de la variable recherchée :} \[ P_{\text{élec}} = \frac{P_{\text{méc}}}{\eta} \]
    \item \textbf{Résolution numérique :} \[ P_{\text{élec}} = \frac{\Val{(\vsi{m}*9.81*\vsi{v})}}{\num{\vsi{eta}}} = \Res{(\vsi{m}*9.81*\vsi{v}/\vsi{eta})}{W} \]
    \item \textbf{Réponse :} $P_{\text{élec}} = \mathbf{\Res{(\vsi{m}*9.81*\vsi{v}/\vsi{eta})}{W}}$
\end{enumerate}
