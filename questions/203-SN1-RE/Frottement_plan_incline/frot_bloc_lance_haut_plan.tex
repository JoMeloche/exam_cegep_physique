\begin{enumerate}[label=\alph*)]
    \item Quel est le module de l'accélération du bloc pendant la montée ?
    \item Quelle distance le bloc parcourt-il le long du plan avant de s'arrêter ?
    \item Quel est le module de l'accélération pendant la descente ?
    \item Quelle est la vitesse du bloc lorsqu'il repasse à son point de départ ?
\end{enumerate}
