\subsubsection*{Question a) Vitesse commune}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m_1 = \qty{\v{m1}}{kg}$, $v_1 = \qty{\v{v1}}{m/s}$, $m_2 = \qty{\v{m2}}{kg}$, $v_2 = \qty{0}{m/s}$ ; variable recherchée : $v_f$
    \item \textbf{Équations pertinentes :} conservation de la quantité de mouvement (collision parfaitement inélastique) : \[ m_1 v_1 + m_2 \cdot 0 = (m_1 + m_2) v_f \]
    \item \textbf{Isolation de la variable recherchée :} \[ v_f = \frac{m_1 v_1}{m_1 + m_2} \]
    \item \textbf{Résolution numérique :} \[ v_f = \frac{\num{\v{m1}} \cdot \num{\v{v1}}}{\num{\v{m1}} + \num{\v{m2}}} = \Res{(\vsi{m1}*\vsi{v1}/(\vsi{m1} + \vsi{m2}))}{m/s} \]
    \item \textbf{Réponse :} $v_f = \mathbf{\Res{(\vsi{m1}*\vsi{v1}/(\vsi{m1} + \vsi{m2}))}{m/s}}$ dans le sens initial du wagon 1.
\end{enumerate}

\subsubsection*{Question b) Énergie cinétique perdue}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m_1$, $m_2$, $v_1$, $v_f = \Res{(\vsi{m1}*\vsi{v1}/(\vsi{m1} + \vsi{m2}))}{m/s}$ ; variable recherchée : $|\Delta K|$
    \item \textbf{Équations pertinentes :} \[ K_i = \tfrac{1}{2}m_1 v_1^2 \qquad K_f = \tfrac{1}{2}(m_1 + m_2)v_f^2 \]
    \item \textbf{Isolation de la variable recherchée :} \[ |\Delta K| = K_i - K_f \]
    \item \textbf{Résolution numérique :} \[ K_i = \Res{0.5*\vsi{m1}*\vsi{v1}^2}{J} \qquad K_f = \Res{0.5*(\vsi{m1} + \vsi{m2})*((\vsi{m1}*\vsi{v1}/(\vsi{m1} + \vsi{m2})))^2}{J} \qquad |\Delta K| = \Res{(0.5*\vsi{m1}*\vsi{v1}^2 - 0.5*(\vsi{m1} + \vsi{m2})*((\vsi{m1}*\vsi{v1}/(\vsi{m1} + \vsi{m2})))^2)}{J} \]
    \item \textbf{Réponse :} $|\Delta K| = \mathbf{\Res{(0.5*\vsi{m1}*\vsi{v1}^2 - 0.5*(\vsi{m1} + \vsi{m2})*((\vsi{m1}*\vsi{v1}/(\vsi{m1} + \vsi{m2})))^2)}{J}}$ (transformée en chaleur, son et déformation).
\end{enumerate}

\subsubsection*{Question c) Pourcentage perdu}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $K_i = \Res{0.5*\vsi{m1}*\vsi{v1}^2}{J}$, $|\Delta K| = \Res{(0.5*\vsi{m1}*\vsi{v1}^2 - 0.5*(\vsi{m1} + \vsi{m2})*((\vsi{m1}*\vsi{v1}/(\vsi{m1} + \vsi{m2})))^2)}{J}$ ; variable recherchée : le pourcentage
    \item \textbf{Équations pertinentes :} \[ \% = \frac{|\Delta K|}{K_i} \times 100 \]
    \item \textbf{Isolation de la variable recherchée :} en remplaçant $v_f$ : \[ \% = \frac{m_2}{m_1 + m_2} \times 100 \]
    \item \textbf{Résolution numérique :} \[ \% = \frac{\num{\v{m2}}}{\num{\v{m1}} + \num{\v{m2}}} \times 100 = \Res{(100*\vsi{m2}/(\vsi{m1} + \vsi{m2}))}{\percent} \]
    \item \textbf{Réponse :} $\mathbf{\Res{(100*\vsi{m2}/(\vsi{m1} + \vsi{m2}))}{\percent}}$ de l'énergie cinétique initiale est perdue.
\end{enumerate}
