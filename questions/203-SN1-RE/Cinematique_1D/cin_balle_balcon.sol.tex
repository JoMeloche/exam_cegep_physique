\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Hauteur maximale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Vitesse initiale vers le haut : $v_0 = \qty{\v{v0}}{m/s}$
        \item Hauteur du balcon : $y_0 = \qty{\v{h}}{m}$
        \item Axe $y$ vers le haut, origine au sol, $a_y = -g = \qty{-9.81}{m/s^2}$
        \item Au sommet : $v_y = \qty{0}{m/s}$
        \item Variable recherchée : $y_{\max}$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v_y^2 = v_{0}^2 + 2 a_y (y - y_0) \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ 0 = v_0^2 - 2g(y_{\max} - y_0) \implies y_{\max} = y_0 + \frac{v_0^2}{2g} \]
    \item \textbf{Résolution numérique :}
    \[ y_{\max} = \num{\vsi{h}} + \frac{(\num{\vsi{v0}})^2}{2 \cdot \num{9.81}} = \Res{\vsi{h} + \vsi{v0}^2/(2*9.81)}{m} \]
    \item \textbf{Réponse :} la balle monte jusqu'à \textbf{\Res{\vsi{h} + \vsi{v0}^2/(2*9.81)}{m}} au-dessus du sol.
\end{enumerate}

\subsubsection*{Question b) Temps de vol}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Vitesse initiale vers le haut : $v_0 = \qty{\v{v0}}{m/s}$
        \item Hauteur du balcon : $y_0 = \qty{\v{h}}{m}$
        \item Au sol : $y = \qty{0}{m}$
        \item Variable recherchée : $t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ y = y_0 + v_0 t - \frac{1}{2} g t^2 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \frac{1}{2} g t^2 - v_0 t - y_0 = 0 \implies t = \frac{v_0 + \sqrt{v_0^2 + 2 g y_0}}{g} \quad \text{(racine positive)} \]
    \item \textbf{Résolution numérique :}
    \[ t = \frac{\num{\vsi{v0}} + \sqrt{(\num{\vsi{v0}})^2 + 2 \cdot \num{9.81} \cdot \num{\vsi{h}}}}{\num{9.81}} = \Res{(\vsi{v0} + sqrt(\vsi{v0}^2 + 2*9.81*\vsi{h}))/9.81}{s} \]
    \item \textbf{Réponse :} la balle touche le sol \textbf{\Res{(\vsi{v0} + sqrt(\vsi{v0}^2 + 2*9.81*\vsi{h}))/9.81}{s}} après le lancer.
\end{enumerate}

\subsubsection*{Question c) Vitesse à l'impact}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Vitesse initiale vers le haut : $v_0 = \qty{\v{v0}}{m/s}$
        \item Hauteur du balcon : $y_0 = \qty{\v{h}}{m}$
        \item Variable recherchée : $v$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v_y^2 = v_0^2 + 2 a_y (y - y_0) = v_0^2 + 2 g y_0 \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ |v_y| = \sqrt{v_0^2 + 2 g y_0} \]
    \item \textbf{Résolution numérique :}
    \[ |v_y| = \sqrt{(\num{\vsi{v0}})^2 + 2 \cdot \num{9.81} \cdot \num{\vsi{h}}} = \Res{sqrt(\vsi{v0}^2 + 2*9.81*\vsi{h})}{m/s} \]
    \item \textbf{Réponse :} la balle frappe le sol à \textbf{\Res{sqrt(\vsi{v0}^2 + 2*9.81*\vsi{h})}{m/s}}, vers le bas.
\end{enumerate}
