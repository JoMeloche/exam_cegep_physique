\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Accélération}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Masse 1 : $m_1 = \qty{\v{m1}}{kg}$
        \item Masse 2 : $m_2 = \qty{\v{m2}}{kg}$
        \item Bloc 1 (plus lourd) descend, bloc 2 monte ; même $a$ et même $T$
        \item Variable recherchée : $a$ (m/s$^2$)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ \text{Bloc 1 : } m_1 g - T = m_1 a \qquad \text{Bloc 2 : } T - m_2 g = m_2 a \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \text{En additionnant : } (m_1 - m_2) g = (m_1 + m_2) a \implies a = \frac{(m_1 - m_2) g}{m_1 + m_2} \]
    \item \textbf{Résolution numérique :}
    \[ a = \frac{(\num{\vsi{m1}} - \num{\vsi{m2}})\cdot\num{9.81}}{\num{\vsi{m1}} + \num{\vsi{m2}}} = \Res{(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})}{m/s^2} \]
    \item \textbf{Réponse :} les blocs accélèrent à \textbf{\Res{(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})}{m/s^2}}.
\end{enumerate}

\subsubsection*{Question b) Tension}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item Masse 2 : $m_2 = \qty{\v{m2}}{kg}$
        \item $a = \Res{(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})}{m/s^2}$
        \item Variable recherchée : $T$ (N)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ T - m_2 g = m_2 a \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ T = m_2 (g + a) \]
    \item \textbf{Résolution numérique :}
    \[ T = \num{\vsi{m2}} \cdot (\num{9.81} + \Val{(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})}) = \Res{2*\vsi{m1}*\vsi{m2}*9.81/(\vsi{m1} + \vsi{m2})}{N} \]
    \item \textbf{Réponse :} la tension vaut \textbf{\Res{2*\vsi{m1}*\vsi{m2}*9.81/(\vsi{m1} + \vsi{m2})}{N}}.
\end{enumerate}

\subsubsection*{Question c) Vitesse après la descente}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $a = \Res{(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})}{m/s^2}$
        \item Distance de descente : $\Delta y = \qty{\v{d}}{m}$
        \item $v_0 = 0$
        \item Variable recherchée : $v$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :}
    \[ v^2 = v_0^2 + 2 a \Delta y \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ v = \sqrt{2 a \Delta y} \]
    \item \textbf{Résolution numérique :}
    \[ v = \sqrt{2 \cdot \Val{(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})} \cdot \num{\vsi{d}}} = \Res{sqrt(2*(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})*\vsi{d})}{m/s} \]
    \item \textbf{Réponse :} les blocs vont à \textbf{\Res{sqrt(2*(\vsi{m1} - \vsi{m2})*9.81/(\vsi{m1} + \vsi{m2})*\vsi{d})}{m/s}}.
\end{enumerate}
