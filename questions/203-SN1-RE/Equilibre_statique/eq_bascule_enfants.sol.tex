\subsubsection*{Question a) Position du second enfant}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m_1 = \qty{\v{m1}}{kg}$, $d_1 = \qty{\v{d1}}{m}$, $m_2 = \qty{\v{m2}}{kg}$ ; variable recherchée : $d_2$
    \item \textbf{Équations pertinentes :} moments de force autour du pivot (le poids de la planche et la force du pivot y passent) :\[ \sum \tau = m_1 g d_1 - m_2 g d_2 = 0 \]
    \item \textbf{Isolation de la variable recherchée :} \[ d_2 = \frac{m_1 d_1}{m_2} \]
    \item \textbf{Résolution numérique :} \[ d_2 = \frac{\num{\v{m1}} \cdot \num{\v{d1}}}{\num{\v{m2}}} = \Res{\vsi{m1}*\vsi{d1}/\vsi{m2}}{m} \]
    \item \textbf{Réponse :} $d_2 = \mathbf{\Res{\vsi{m1}*\vsi{d1}/\vsi{m2}}{m}}$
\end{enumerate}

\subsubsection*{Question b) Force du pivot}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m_1$, $m_2$, $M = \qty{\v{M}}{kg}$ ; variable recherchée : $F_p$
    \item \textbf{Équations pertinentes :} \[ \sum F_y = F_p - (m_1 + m_2 + M)\,g = 0 \]
    \item \textbf{Isolation de la variable recherchée :} \[ F_p = (m_1 + m_2 + M)\,g \]
    \item \textbf{Résolution numérique :} \[ F_p = (\num{\v{m1}} + \num{\v{m2}} + \num{\v{M}}) \cdot \num{9.81} = \Res{(\vsi{m1} + \vsi{m2} + \vsi{M})*9.81}{N} \]
    \item \textbf{Réponse :} $F_p = \mathbf{\Res{(\vsi{m1} + \vsi{m2} + \vsi{M})*9.81}{N}}$, vers le haut.
\end{enumerate}

\subsubsection*{Question c) Moment de force résultant}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m_1$, $m_2$, $d_1 = d_2 = \qty{\v{d1}}{m}$ ; variable recherchée : $|\tau|$
    \item \textbf{Équations pertinentes :} \[ \sum \tau = m_1 g d_1 - m_2 g d_1 \]
    \item \textbf{Isolation de la variable recherchée :} \[ |\tau| = (m_1 - m_2)\,g\,d_1 \]
    \item \textbf{Résolution numérique :} \[ |\tau| = (\num{\v{m1}} - \num{\v{m2}}) \cdot \num{9.81} \cdot \num{\v{d1}} = \Res{(\vsi{m1} - \vsi{m2})*9.81*\vsi{d1}}{N.m} \]
    \item \textbf{Réponse :} $|\tau| = \mathbf{\Res{(\vsi{m1} - \vsi{m2})*9.81*\vsi{d1}}{N.m}}$ : la bascule penche du côté de l'enfant 1.
\end{enumerate}
