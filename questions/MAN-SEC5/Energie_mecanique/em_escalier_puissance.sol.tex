\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Énergie potentielle gagnée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $m = \qty{\v{m}}{kg}$, $\Delta h = \qty{\v{h}}{m}$, $g = \qty{9.81}{m/s^2}$
        \item Variable recherchée : $\Delta E_p$ (J)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ \Delta E_p = m g \Delta h \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ \Delta E_p = \num{\v{m}} \cdot \num{9.81} \cdot \num{\v{h}} = \Res{\v{m}*9.81*\v{h}}{J} \]
    \item \textbf{Réponse :} l'élève gagne \textbf{\Res{\v{m}*9.81*\v{h}}{J}} d'énergie potentielle.
\end{enumerate}

\subsubsection*{Question b) Puissance moyenne}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $\Delta E_p = \Res{\v{m}*9.81*\v{h}}{J}$, $\Delta t = \qty{\v{t}}{s}$
        \item Variable recherchée : $P$ (W)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ P = \frac{\Delta E}{\Delta t} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ P = \frac{\Val{\v{m}*9.81*\v{h}}}{\num{\v{t}}} = \Res{\v{m}*9.81*\v{h}/\v{t}}{W} \]
    \item \textbf{Réponse :} l'élève développe une puissance moyenne de \textbf{\Res{\v{m}*9.81*\v{h}/\v{t}}{W}}.
\end{enumerate}

\subsubsection*{Question c) Durée d'allumage de l'ampoule}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $E = \Res{\v{m}*9.81*\v{h}}{J}$, $P_{\text{ampoule}} = \qty{\v{P}}{W}$
        \item Variable recherchée : $\Delta t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ P = \frac{E}{\Delta t} \]
    \item \textbf{Isolation de la variable recherchée :} \[ \Delta t = \frac{E}{P} \]
    \item \textbf{Résolution numérique :}
    \[ \Delta t = \frac{\Val{\v{m}*9.81*\v{h}}}{\num{\v{P}}} = \Res{\v{m}*9.81*\v{h}/\v{P}}{s} \]
    \item \textbf{Réponse :} cette énergie allumerait l'ampoule pendant \textbf{\Res{\v{m}*9.81*\v{h}/\v{P}}{s}}, soit \Res{\v{m}*9.81*\v{h}/\v{P}/60}{min}.
\end{enumerate}
