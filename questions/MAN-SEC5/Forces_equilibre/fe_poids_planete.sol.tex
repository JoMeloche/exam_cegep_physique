\subsubsection*{Question a) Poids sur la Terre}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m = \qty{\v{m}}{kg}$, $g = \qty{9.81}{N/kg}$ ; variable recherchée : $F_g$
    \item \textbf{Équations pertinentes :}
    \[ F_g = mg \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ F_g = \qty{\v{m}}{kg} \cdot \qty{9.81}{N/kg} = \Res{\v{m}*9.81}{N} \]
    \item \textbf{Réponse :} $F_g = \mathbf{\Res{\v{m}*9.81}{N}}$.
\end{enumerate}
\subsubsection*{Question b) Champ gravitationnel de la planète}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $F_g = \qty{\v{W}}{N}$, $m = \qty{\v{m}}{kg}$ ; variable recherchée : $g_p$
    \item \textbf{Équations pertinentes :}
    \[ F_g = m g_p \]
    \item \textbf{Isolation de la variable recherchée :} 
    \[ g_p = \frac{F_g}{m} \]
    \item \textbf{Résolution numérique :}
    \[ g_p = \frac{\qty{\v{W}}{N}}{\qty{\v{m}}{kg}} = \Res{\v{W}/\v{m}}{N/kg} \]
    \item \textbf{Réponse :} $g_p = \mathbf{\Res{\v{W}/\v{m}}{N/kg}}$.
\end{enumerate}
\subsubsection*{Question c) Sur la Lune}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $m = \qty{\v{m}}{kg}$ (la masse ne dépend pas du lieu), $g_L = \qty{1.62}{N/kg}$ ; variable recherchée : $F_{gL}$
    \item \textbf{Équations pertinentes :}
    \[ F_{gL} = m g_L \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ F_{gL} = \qty{\v{m}}{kg} \cdot \qty{1.62}{N/kg} = \Res{\v{m}*1.62}{N} \]
    \item \textbf{Réponse :} la masse reste \textbf{\qty{\v{m}}{kg}} (quantité de matière) ; le poids sur la Lune est de \textbf{\Res{\v{m}*1.62}{N}}.
\end{enumerate}
