\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Durée de la chute}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $\Delta y = \qty{-\v{h}}{m}$ (vers le bas), $v_0 = \qty{0}{m/s}$, $a = -g = \qty{-9.81}{m/s^2}$
        \item Variable recherchée : $t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ \Delta y = v_0 t + \tfrac{1}{2} a t^2 \]
    \item \textbf{Isolation de la variable recherchée :} avec $v_0 = 0$ :
    \[ t = \sqrt{\frac{2\,\Delta y}{a}} \]
    \item \textbf{Résolution numérique :}
    \[ t = \sqrt{\frac{2 \cdot (\num{-\v{h}})}{\num{-9.81}}} = \Res{sqrt(2*\v{h}/9.81)}{s} \]
    \item \textbf{Réponse :} la pierre atteint l'eau après \textbf{\Res{sqrt(2*\v{h}/9.81)}{s}}.
\end{enumerate}

\subsubsection*{Question b) Vitesse à l'impact}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_0 = \qty{0}{m/s}$, $a = \qty{-9.81}{m/s^2}$, $\Delta y = \qty{-\v{h}}{m}$
        \item Variable recherchée : $v$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ v^2 = v_0^2 + 2 a \Delta y \]
    \item \textbf{Isolation de la variable recherchée :} \[ |v| = \sqrt{2 g h} \]
    \item \textbf{Résolution numérique :}
    \[ |v| = \sqrt{2 \cdot \num{9.81} \cdot \num{\v{h}}} = \Res{sqrt(2*9.81*\v{h})}{m/s} \]
    \item \textbf{Réponse :} la pierre frappe l'eau à \textbf{\Res{sqrt(2*9.81*\v{h})}{m/s}}, vers le bas.
\end{enumerate}

\subsubsection*{Question c) Hauteur après \qty{\v{t1}}{s}}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $t = \qty{\v{t1}}{s}$, $v_0 = \qty{0}{m/s}$, $a = \qty{-9.81}{m/s^2}$, $y_0 = \qty{\v{h}}{m}$ (origine à la surface de l'eau)
        \item Variable recherchée : $y$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ y = y_0 + v_0 t + \tfrac{1}{2} a t^2 \]
    \item \textbf{Isolation de la variable recherchée :} \[ y = h - \tfrac{1}{2} g t^2 \]
    \item \textbf{Résolution numérique :}
    \[ y = \num{\v{h}} - \tfrac{1}{2} \cdot \num{9.81} \cdot (\num{\v{t1}})^2 = \Res{\v{h} - 0.5*9.81*\v{t1}^2}{m} \]
    \item \textbf{Réponse :} la pierre est à \textbf{\Res{\v{h} - 0.5*9.81*\v{t1}^2}{m}} au-dessus de l'eau.
\end{enumerate}

\subsubsection*{Question d) Instant où le plouf est entendu}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $t_{\text{chute}} = \Res{sqrt(2*\v{h}/9.81)}{s}$, $h = \qty{\v{h}}{m}$, $v_{\text{son}} = \qty{340}{m/s}$
        \item Variable recherchée : $t_{\text{total}}$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} le son remonte en MRU :
    \[ t_{\text{son}} = \frac{h}{v_{\text{son}}} \qquad t_{\text{total}} = t_{\text{chute}} + t_{\text{son}} \]
    \item \textbf{Isolation de la variable recherchée :} les équations sont déjà isolées.
    \item \textbf{Résolution numérique :}
    \[ t_{\text{total}} = \Val{sqrt(2*\v{h}/9.81)}\,\unit{s} + \frac{\qty{\v{h}}{m}}{\qty{340}{m/s}} = \Res{sqrt(2*\v{h}/9.81) + \v{h}/340}{s} \]
    \item \textbf{Réponse :} l'enfant entend le plouf \textbf{\Res{sqrt(2*\v{h}/9.81) + \v{h}/340}{s}} après avoir lâché la pierre.
\end{enumerate}
