\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Temps de montée}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_0 = \qty{\v{v0}}{m/s}$, au sommet $v = \qty{0}{m/s}$, $a = \qty{-9.81}{m/s^2}$
        \item Variable recherchée : $t_m$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ v = v_0 + a t \]
    \item \textbf{Isolation de la variable recherchée :} \[ t_m = \frac{v - v_0}{a} = \frac{v_0}{g} \]
    \item \textbf{Résolution numérique :}
    \[ t_m = \frac{\num{\v{v0}}}{\num{9.81}} = \Res{\v{v0}/9.81}{s} \]
    \item \textbf{Réponse :} le ballon atteint le sommet après \textbf{\Res{\v{v0}/9.81}{s}}.
\end{enumerate}

\subsubsection*{Question b) Hauteur maximale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_0 = \qty{\v{v0}}{m/s}$, $v = \qty{0}{m/s}$, $a = \qty{-9.81}{m/s^2}$
        \item Variable recherchée : $h_{\max}$ (m)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ v^2 = v_0^2 + 2 a \Delta y \]
    \item \textbf{Isolation de la variable recherchée :} \[ h_{\max} = \frac{v_0^2}{2g} \]
    \item \textbf{Résolution numérique :}
    \[ h_{\max} = \frac{(\num{\v{v0}})^2}{2 \cdot \num{9.81}} = \Res{\v{v0}^2/(2*9.81)}{m} \]
    \item \textbf{Réponse :} le ballon monte à \textbf{\Res{\v{v0}^2/(2*9.81)}{m}} au-dessus de ses mains.
\end{enumerate}

\subsubsection*{Question c) Durée totale du vol}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_0 = \qty{\v{v0}}{m/s}$, retour à la position de départ : $\Delta y = \qty{0}{m}$
        \item Variable recherchée : $t$ (s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ \Delta y = v_0 t - \tfrac{1}{2} g t^2 = 0 \]
    \item \textbf{Isolation de la variable recherchée :} en écartant la solution $t = 0$ :
    \[ t = \frac{2 v_0}{g} \]
    \item \textbf{Résolution numérique :}
    \[ t = \frac{2 \cdot \num{\v{v0}}}{\num{9.81}} = \Res{2*\v{v0}/9.81}{s} \]
    \item \textbf{Réponse :} le ballon revient à la hauteur de ses mains après \textbf{\Res{2*\v{v0}/9.81}{s}} (deux fois le temps de montée).
\end{enumerate}

\subsubsection*{Question d) Vitesse à \qty{\v{h1}}{m}}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_0 = \qty{\v{v0}}{m/s}$, $\Delta y = \qty{\v{h1}}{m}$, $a = \qty{-9.81}{m/s^2}$
        \item Variable recherchée : $|v|$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ v^2 = v_0^2 + 2 a \Delta y \]
    \item \textbf{Isolation de la variable recherchée :} \[ |v| = \sqrt{v_0^2 - 2 g \Delta y} \]
    \item \textbf{Résolution numérique :}
    \[ |v| = \sqrt{(\num{\v{v0}})^2 - 2 \cdot \num{9.81} \cdot \num{\v{h1}}} = \Res{sqrt(\v{v0}^2 - 2*9.81*\v{h1})}{m/s} \]
    \item \textbf{Réponse :} le ballon passe à cette hauteur à \textbf{\Res{sqrt(\v{v0}^2 - 2*9.81*\v{h1})}{m/s}}, vers le haut à la montée et vers le bas à la descente.
\end{enumerate}
