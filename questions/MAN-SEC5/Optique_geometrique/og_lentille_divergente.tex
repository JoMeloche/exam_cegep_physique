\begin{enumerate}[label=\alph*)]
    \item À quelle distance de la lentille l'image se forme-t-elle ? De quel côté de la lentille ?
    \item Déterminez le grandissement.
    \item Déterminez la hauteur de l'image et décrivez ses caractéristiques.
\end{enumerate}
