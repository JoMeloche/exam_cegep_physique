\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\providecommand{\ResSci}[2]{\qty[round-mode=figures,round-precision=3,exponent-mode=scientific]{\fpeval{#1}}{#2}}
\subsubsection*{Question a) Position de l'image}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f = \qty{\v{f}}{cm}$, $d_o = \qty{\v{dob}}{cm}$ ; variable recherchée : $d_i$
    \item \textbf{Équations pertinentes :} équation des lentilles minces :
    \[ \frac{1}{f} = \frac{1}{d_o} + \frac{1}{d_i} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ d_i = \frac{1}{\frac{1}{f} - \frac{1}{d_o}} \]
    \item \textbf{Résolution numérique :}
    \[ d_i = \frac{1}{\frac{1}{\qty{\v{f}}{cm}} - \frac{1}{\qty{\v{dob}}{cm}}} = \Res{1/(1/\v{f} - 1/\v{dob})}{cm} \]
    \item \textbf{Réponse :} l'image se forme à \textbf{\Res{1/(1/\v{f} - 1/\v{dob})}{cm}} de la lentille, du côté opposé à l'objet ($d_i > 0$).
\end{enumerate}

\subsubsection*{Question b) Grandissement}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $d_i = \Res{1/(1/\v{f} - 1/\v{dob})}{cm}$, $d_o = \qty{\v{dob}}{cm}$ ; variable recherchée : $G$
    \item \textbf{Équations pertinentes :}
    \[ G = \frac{h_i}{h_o} = -\frac{d_i}{d_o} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ G = -\frac{\Val{1/(1/\v{f} - 1/\v{dob})}\,\unit{cm}}{\qty{\v{dob}}{cm}} = \num[round-mode=figures,round-precision=3]{\fpeval{-(1/(1/\v{f} - 1/\v{dob}))/\v{dob}}} \]
    \item \textbf{Réponse :} $G = \mathbf{\num[round-mode=figures,round-precision=3]{\fpeval{-(1/(1/\v{f} - 1/\v{dob}))/\v{dob}}}}$.
\end{enumerate}

\subsubsection*{Question c) Hauteur et caractéristiques de l'image}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $G = \num[round-mode=figures,round-precision=3]{\fpeval{-(1/(1/\v{f} - 1/\v{dob}))/\v{dob}}}$, $h_o = \qty{\v{ho}}{cm}$ ; variable recherchée : $h_i$
    \item \textbf{Équations pertinentes :}
    \[ G = \frac{h_i}{h_o} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ h_i = G\,h_o \]
    \item \textbf{Résolution numérique :}
    \[ h_i = \Val{-(1/(1/\v{f} - 1/\v{dob}))/\v{dob}} \cdot \qty{\v{ho}}{cm} = \Res{-\v{ho}*(1/(1/\v{f} - 1/\v{dob}))/\v{dob}}{cm} \]
    \item \textbf{Réponse :} $h_i = \mathbf{\Res{-\v{ho}*(1/(1/\v{f} - 1/\v{dob}))/\v{dob}}{cm}}$. L'image est \textbf{réelle} ($d_i > 0$) et \textbf{renversée} ($G < 0$) ; elle est agrandie si $|G| > 1$ et réduite si $|G| < 1$ (ici $|G| = \Val{(1/(1/\v{f} - 1/\v{dob}))/\v{dob}}$).
\end{enumerate}
