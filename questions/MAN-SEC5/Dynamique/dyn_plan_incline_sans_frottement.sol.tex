\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\subsubsection*{Question a) Composante parallèle du poids}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $m = \qty{\v{m}}{kg}$, $\theta = \ang{\v{th}}$, $g = \qty{9.81}{m/s^2}$
        \item Forces : poids $m\vec g$ (vertical, vers le bas) et normale $\vec F_N$ (perpendiculaire à la rampe)
        \item Variable recherchée : $F_{g\parallel}$ (N)
    \end{itemize}
    \item \textbf{Équations pertinentes :} axe $x$ parallèle à la rampe (vers le bas) :
    \[ F_{g\parallel} = m g \sin\theta \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ F_{g\parallel} = \num{\v{m}} \cdot \num{9.81} \cdot \sin(\ang{\v{th}}) = \Res{\v{m}*9.81*sind(\v{th})}{N} \]
    \item \textbf{Réponse :} la composante parallèle du poids est de \textbf{\Res{\v{m}*9.81*sind(\v{th})}{N}}, vers le bas de la rampe.
\end{enumerate}

\subsubsection*{Question b) Force normale}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $m = \qty{\v{m}}{kg}$, $\theta = \ang{\v{th}}$
        \item Variable recherchée : $F_N$ (N)
    \end{itemize}
    \item \textbf{Équations pertinentes :} aucune accélération perpendiculaire à la rampe :
    \[ \sum F_y = F_N - m g \cos\theta = 0 \]
    \item \textbf{Isolation de la variable recherchée :} \[ F_N = m g \cos\theta \]
    \item \textbf{Résolution numérique :}
    \[ F_N = \num{\v{m}} \cdot \num{9.81} \cdot \cos(\ang{\v{th}}) = \Res{\v{m}*9.81*cosd(\v{th})}{N} \]
    \item \textbf{Réponse :} la force normale est de \textbf{\Res{\v{m}*9.81*cosd(\v{th})}{N}}.
\end{enumerate}

\subsubsection*{Question c) Accélération}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $F_{g\parallel} = \Res{\v{m}*9.81*sind(\v{th})}{N}$, $m = \qty{\v{m}}{kg}$
        \item Variable recherchée : $a$ (m/s²)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ \sum F_x = m g \sin\theta = m a \]
    \item \textbf{Isolation de la variable recherchée :} \[ a = g \sin\theta \]
    \item \textbf{Résolution numérique :}
    \[ a = \num{9.81} \cdot \sin(\ang{\v{th}}) = \Res{9.81*sind(\v{th})}{m/s^2} \]
    \item \textbf{Réponse :} la caisse accélère à \textbf{\Res{9.81*sind(\v{th})}{m/s^2}} le long de la rampe (indépendamment de sa masse).
\end{enumerate}

\subsubsection*{Question d) Vitesse au bas de la rampe}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :}
    \begin{itemize}[label= -]
        \item $v_0 = \qty{0}{m/s}$, $a = \Res{9.81*sind(\v{th})}{m/s^2}$, $d = \qty{\v{L}}{m}$
        \item Variable recherchée : $v$ (m/s)
    \end{itemize}
    \item \textbf{Équations pertinentes :} \[ v^2 = v_0^2 + 2 a d \]
    \item \textbf{Isolation de la variable recherchée :} \[ v = \sqrt{2 a d} \]
    \item \textbf{Résolution numérique :}
    \[ v = \sqrt{2 \cdot \Val{9.81*sind(\v{th})} \cdot \num{\v{L}}} = \Res{sqrt(2*9.81*sind(\v{th})*\v{L})}{m/s} \]
    \item \textbf{Réponse :} la caisse arrive au bas de la rampe à \textbf{\Res{sqrt(2*9.81*sind(\v{th})*\v{L})}{m/s}}.
\end{enumerate}
