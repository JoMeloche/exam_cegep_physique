\subsubsection*{Question a) Longueurs d'onde}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f_1 = \qty{\v{f1}}{Hz}$, $f_2 = \qty{\v{f2}}{Hz}$, $v = \qty{\v{v}}{m/s}$ ; variables recherchées : $\lambda_1$, $\lambda_2$
    \item \textbf{Équations pertinentes :}
    \[ v = \lambda f \]
    \item \textbf{Isolation de la variable recherchée :} 
    \[ \lambda = \frac{v}{f} \]
    \item \textbf{Résolution numérique :}
    \[ \lambda_1 = \frac{\qty{\v{v}}{m/s}}{\qty{\v{f1}}{Hz}} = \Res{\v{v}/\v{f1}}{m} \qquad \lambda_2 = \frac{\qty{\v{v}}{m/s}}{\qty{\v{f2}}{Hz}} = \Res{\v{v}/\v{f2}}{m} = \Res{100*\v{v}/\v{f2}}{cm} \]
    \item \textbf{Réponse :} $\lambda_1 = \mathbf{\Res{\v{v}/\v{f1}}{m}}$ et $\lambda_2 = \mathbf{\Res{100*\v{v}/\v{f2}}{cm}}$.
\end{enumerate}
\subsubsection*{Question b) Rapport des longueurs d'onde}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $\lambda_1 = \Res{\v{v}/\v{f1}}{m}$, $\lambda_2 = \Res{\v{v}/\v{f2}}{m}$ ; variable recherchée : $\lambda_1/\lambda_2$
    \item \textbf{Équations pertinentes :}
    \[ \frac{\lambda_1}{\lambda_2} = \frac{v/f_1}{v/f_2} = \frac{f_2}{f_1} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ \frac{\lambda_1}{\lambda_2} = \frac{\qty{\v{f2}}{Hz}}{\qty{\v{f1}}{Hz}} = \num[round-mode=figures,round-precision=3]{\fpeval{\v{f2}/\v{f1}}} \]
    \item \textbf{Réponse :} la longueur d'onde de la note grave est \textbf{\num[round-mode=figures,round-precision=3]{\fpeval{\v{f2}/\v{f1}}}} fois plus grande. Les deux sons voyagent à la même vitesse dans l'air : ils arrivent en même temps.
\end{enumerate}
\subsubsection*{Question c) Période de la note grave}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f_1 = \qty{\v{f1}}{Hz}$ ; variable recherchée : $T_1$
    \item \textbf{Équations pertinentes :}
    \[ T = \frac{1}{f} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ T_1 = \frac{1}{\qty{\v{f1}}{Hz}} = \Res{1/\v{f1}}{s} = \Res{1000/\v{f1}}{ms} \]
    \item \textbf{Réponse :} $T_1 = \mathbf{\Res{1000/\v{f1}}{ms}}$.
\end{enumerate}
