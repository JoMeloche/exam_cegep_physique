\subsubsection*{Question a) Fréquence}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $\lambda = \qty{\v{lam}}{nm} = \qty{\vsi{lam}}{m}$, $c = \qty{3.00e8}{m/s}$ ; variable recherchée : $f$
    \item \textbf{Équations pertinentes :}
    \[ c = \lambda f \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ f = \frac{c}{\lambda} \]
    \item \textbf{Résolution numérique :}
    \[ f = \frac{\qty{3.00e8}{m/s}}{\qty{\v{lam}e-9}{m}} = \ResSci{3e8/\vsi{lam}}{Hz} \]
    \item \textbf{Réponse :} $f = \mathbf{\ResSci{3e8/\vsi{lam}}{Hz}}$.
\end{enumerate}

\subsubsection*{Question b) Vitesse dans le milieu}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $c = \qty{3.00e8}{m/s}$, $n = \num{\v{n}}$ ; variable recherchée : $v$
    \item \textbf{Équations pertinentes :}
    \[ n = \frac{c}{v} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ v = \frac{c}{n} \]
    \item \textbf{Résolution numérique :}
    \[ v = \frac{\qty{3.00e8}{m/s}}{\num{\v{n}}} = \ResSci{3e8/\v{n}}{m/s} \]
    \item \textbf{Réponse :} $v = \mathbf{\ResSci{3e8/\v{n}}{m/s}}$.
\end{enumerate}

\subsubsection*{Question c) Longueur d'onde dans le milieu}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $v = \ResSci{3e8/\v{n}}{m/s}$, $f = \ResSci{3e8/\vsi{lam}}{Hz}$ (la fréquence ne change pas) ; variable recherchée : $\lambda_n$
    \item \textbf{Équations pertinentes :}
    \[ v = \lambda_n f \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \lambda_n = \frac{v}{f} = \frac{\lambda}{n} \]
    \item \textbf{Résolution numérique :}
    \[ \lambda_n = \frac{\qty{\v{lam}}{nm}}{\num{\v{n}}} = \Res{\v{lam}/\v{n}}{nm} \]
    \item \textbf{Réponse :} $\lambda_n = \mathbf{\Res{\v{lam}/\v{n}}{nm}}$. La couleur ne change pas : elle est déterminée par la fréquence, qui reste la même quand la lumière change de milieu.
\end{enumerate}
