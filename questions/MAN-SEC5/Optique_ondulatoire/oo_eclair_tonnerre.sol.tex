\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\providecommand{\ResSci}[2]{\qty[round-mode=figures,round-precision=3,exponent-mode=scientific]{\fpeval{#1}}{#2}}
\subsubsection*{Question a) Distance de l'éclair}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $v = \qty{\v{v}}{m/s}$, $\Delta t = \qty{\v{t}}{s}$ ; variable recherchée : $d$
    \item \textbf{Équations pertinentes :}
    \[ v = \frac{d}{\Delta t} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ d = v\,\Delta t \]
    \item \textbf{Résolution numérique :}
    \[ d = \qty{\v{v}}{m/s} \cdot \qty{\v{t}}{s} = \Res{\v{v}*\v{t}}{m} \]
    \item \textbf{Réponse :} l'éclair s'est produit à \textbf{\Res{\v{v}*\v{t}}{m}}.
\end{enumerate}

\subsubsection*{Question b) Trajet de la lumière}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $d = \Res{\v{v}*\v{t}}{m}$, $c = \qty{3.00e8}{m/s}$ ; variable recherchée : $t_{\text{lum}}$
    \item \textbf{Équations pertinentes :}
    \[ c = \frac{d}{t_{\text{lum}}} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ t_{\text{lum}} = \frac{d}{c} \]
    \item \textbf{Résolution numérique :}
    \[ t_{\text{lum}} = \frac{\Val{\v{v}*\v{t}}\,\unit{m}}{\qty{3.00e8}{m/s}} = \ResSci{\v{v}*\v{t}/3e8}{s} = \Res{1e6*\v{v}*\v{t}/3e8}{\micro\second} \]
    \item \textbf{Réponse :} $t_{\text{lum}} = \mathbf{\Res{1e6*\v{v}*\v{t}/3e8}{\micro\second}}$, ce qui est négligeable devant \qty{\v{t}}{s} : l'approximation est justifiée.
\end{enumerate}

\subsubsection*{Question c) Longueur d'onde du grondement}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f = \qty{\v{f}}{Hz}$, $v = \qty{\v{v}}{m/s}$ ; variable recherchée : $\lambda$
    \item \textbf{Équations pertinentes :}
    \[ v = \lambda f \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \lambda = \frac{v}{f} \]
    \item \textbf{Résolution numérique :}
    \[ \lambda = \frac{\qty{\v{v}}{m/s}}{\qty{\v{f}}{Hz}} = \Res{\v{v}/\v{f}}{m} \]
    \item \textbf{Réponse :} $\lambda = \mathbf{\Res{\v{v}/\v{f}}{m}}$.
\end{enumerate}
