\providecommand{\Res}[2]{\qty[round-mode=figures,round-precision=3]{\fpeval{#1}}{#2}}
\providecommand{\Val}[1]{\num[round-mode=figures,round-precision=4]{\fpeval{#1}}}
\providecommand{\ResSci}[2]{\qty[round-mode=figures,round-precision=3,exponent-mode=scientific]{\fpeval{#1}}{#2}}
\subsubsection*{Question a) Longueur d'onde}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f = \qty{\v{f}}{MHz} = \qty{\vsi{f}}{Hz}$, $c = \qty{3.00e8}{m/s}$ ; variable recherchée : $\lambda$
    \item \textbf{Équations pertinentes :}
    \[ c = \lambda f \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \lambda = \frac{c}{f} \]
    \item \textbf{Résolution numérique :}
    \[ \lambda = \frac{\qty{3.00e8}{m/s}}{\qty{\vsi{f}}{Hz}} = \Res{3e8/\vsi{f}}{m} \]
    \item \textbf{Réponse :} $\lambda = \mathbf{\Res{3e8/\vsi{f}}{m}}$.
\end{enumerate}

\subsubsection*{Question b) Période}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f = \qty{\vsi{f}}{Hz}$ ; variable recherchée : $T$
    \item \textbf{Équations pertinentes :}
    \[ T = \frac{1}{f} \]
    \item \textbf{Isolation de la variable recherchée :} l'équation est déjà isolée.
    \item \textbf{Résolution numérique :}
    \[ T = \frac{1}{\qty{\vsi{f}}{Hz}} = \ResSci{1/\vsi{f}}{s} = \Res{1e9/\vsi{f}}{ns} \]
    \item \textbf{Réponse :} $T = \mathbf{\Res{1e9/\vsi{f}}{ns}}$.
\end{enumerate}

\subsubsection*{Question c) Durée du trajet}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $d = \qty{\v{d}}{km} = \qty{\vsi{d}}{m}$, $c = \qty{3.00e8}{m/s}$ ; variable recherchée : $t$
    \item \textbf{Équations pertinentes :} vitesse constante :
    \[ c = \frac{d}{t} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ t = \frac{d}{c} \]
    \item \textbf{Résolution numérique :}
    \[ t = \frac{\qty{\vsi{d}}{m}}{\qty{3.00e8}{m/s}} = \ResSci{\vsi{d}/3e8}{s} = \Res{1e6*\vsi{d}/3e8}{\micro\second} \]
    \item \textbf{Réponse :} le signal arrive après \textbf{\Res{1e6*\vsi{d}/3e8}{\micro\second}}, pratiquement instantanément.
\end{enumerate}
