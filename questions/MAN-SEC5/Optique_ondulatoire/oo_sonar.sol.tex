\subsubsection*{Question a) Profondeur}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $v_{\text{eau}} = \qty{\v{vw}}{m/s}$, $\Delta t = \qty{\v{t}}{s}$ (aller-retour) ; variable recherchée : $d$
    \item \textbf{Équations pertinentes :} le son parcourt deux fois la profondeur :
    \[ v_{\text{eau}} = \frac{2d}{\Delta t} \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ d = \frac{v_{\text{eau}}\,\Delta t}{2} \]
    \item \textbf{Résolution numérique :}
    \[ d = \frac{\qty{\v{vw}}{m/s} \cdot \qty{\v{t}}{s}}{2} = \Res{\v{vw}*\v{t}/2}{m} \]
    \item \textbf{Réponse :} le lac a \textbf{\Res{\v{vw}*\v{t}/2}{m}} de profondeur.
\end{enumerate}

\subsubsection*{Question b) Longueur d'onde dans l'eau}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f = \qty{\v{f}}{kHz} = \qty{\vsi{f}}{Hz}$, $v_{\text{eau}} = \qty{\v{vw}}{m/s}$ ; variable recherchée : $\lambda_{\text{eau}}$
    \item \textbf{Équations pertinentes :}
    \[ v = \lambda f \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \lambda_{\text{eau}} = \frac{v_{\text{eau}}}{f} \]
    \item \textbf{Résolution numérique :}
    \[ \lambda_{\text{eau}} = \frac{\qty{\v{vw}}{m/s}}{\qty{\vsi{f}}{Hz}} = \Res{\v{vw}/\vsi{f}}{m} = \Res{1000*\v{vw}/\vsi{f}}{mm} \]
    \item \textbf{Réponse :} $\lambda_{\text{eau}} = \mathbf{\Res{1000*\v{vw}/\vsi{f}}{mm}}$.
\end{enumerate}

\subsubsection*{Question c) Longueur d'onde dans l'air}
\begin{enumerate}[label=\arabic*)]
    \item \textbf{Variables connues et recherchées :} $f = \qty{\vsi{f}}{Hz}$ (inchangée), $v_{\text{air}} = \qty{\v{va}}{m/s}$ ; variable recherchée : $\lambda_{\text{air}}$
    \item \textbf{Équations pertinentes :}
    \[ v = \lambda f \]
    \item \textbf{Isolation de la variable recherchée :}
    \[ \lambda_{\text{air}} = \frac{v_{\text{air}}}{f} \]
    \item \textbf{Résolution numérique :}
    \[ \lambda_{\text{air}} = \frac{\qty{\v{va}}{m/s}}{\qty{\vsi{f}}{Hz}} = \Res{1000*\v{va}/\vsi{f}}{mm} \]
    \item \textbf{Réponse :} $\lambda_{\text{air}} = \mathbf{\Res{1000*\v{va}/\vsi{f}}{mm}}$. La \textbf{fréquence} reste la même (elle est fixée par la source) ; la vitesse et la longueur d'onde changent.
\end{enumerate}
